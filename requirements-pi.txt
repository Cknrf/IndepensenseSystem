# Raspberry Pi-only runtime dependencies. Install on the Pi with:
#   pip install -r requirements.txt -r requirements-pi.txt
pyserial>=3.5
smbus2>=0.4

# Physical buttons (and later buzzer + vibration motor) via GPIO.
# `lgpio` is the underlying kernel-based backend; gpiozero picks it up
# automatically on Pi 5 running the modern gpiochip interface.
gpiozero>=2.0
lgpio>=0.2

# picamera2 is NOT installable via pip — it depends on system libcamera bindings.
# Install via apt:  sudo apt install -y python3-picamera2
# Then create the venv with --system-site-packages so it can be imported.

# sounddevice + soundfile (in requirements.txt) wrap system C libraries.
# The Python packages install fine via pip, but the underlying C libraries
# must be present:  sudo apt install -y libportaudio2 libsndfile1

# YOLOv8 object detection.
#
# Torch is pinned because recent torch wheels (~2.10+) ship with hard
# dependencies on NVIDIA CUDA libraries (nvidia-cublas, nvidia-cudnn, ...) even
# on aarch64 Linux. On a Pi 5 (no NVIDIA GPU) those libraries are ~3 GB of dead
# weight. 2.6.0 is the oldest aarch64 cp313 wheel available and should predate
# the CUDA-deps-as-hard-requirements packaging change.
torch==2.6.0
torchvision==0.21.0
ultralytics>=8.0

# Tagalog text-to-speech: Meta MMS-TTS (`facebook/mms-tts-tgl`), a VITS
# checkpoint run through transformers' `VitsModel`. It rides on the torch
# install above rather than adding a second inference runtime, which is why
# this is the cheap option — transformers is the only new package.
#
# The weights are downloaded separately (see docs/voice.md) and loaded from
# a local directory, so nothing here reaches the network at runtime.
transformers>=4.33

# Tesseract OCR — printed-text reader for the `vision.read` intent.
# The binary + language packs come from apt:
#   sudo apt install -y tesseract-ocr tesseract-ocr-tgl
# `pytesseract` is just the Python wrapper; it shells out to the binary.
pytesseract>=0.3
